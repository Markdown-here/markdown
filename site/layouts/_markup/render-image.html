Any other repo image is loaded from GitHub, so it still shows. */ -}} {{- $src := .Destination -}} {{- $u := urls.Parse $src -}} {{- $source := .PageInner.Params.source -}} {{- if and $source (not $u.IsAbs) (not (hasPrefix $src "/")) -}} {{- $repoPath := path.Join (path.Dir $source) $u.Path -}} {{- if hasPrefix $repoPath "media/" -}} {{- $src = printf "/%s" $repoPath | relURL -}} {{- else -}} {{- $src = printf "%s/raw/%s/%s" site.Params.repo site.Params.branch $repoPath -}} {{- end -}} {{- end -}} {{- /* Marp background images, like ![bg right:30%](...), have settings, not alt text */ -}} {{- $alt := cond (findRE `^bg( |$)` .PlainText 1) "" .PlainText -}} {{- /* no trailing newline */ -}}
