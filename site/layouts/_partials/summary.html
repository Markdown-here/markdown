{{- /* The first paragraph under `## Summary` (or `## Bio` for people) in a reference entry, on one line. Empty if there isn't one. */ -}} {{- $summary := "" -}} {{- with findRESubmatch `(?s)\n## (?:Summary|Bio)\n+(.+?)(?:\n\n|\n#|\z)` (printf "\n%s" .) 1 -}} {{- $summary = index (index . 0) 1 | replaceRE `\s*\n\s*` " " | strings.TrimSpace -}} {{- if or (eq $summary "-") (hasPrefix $summary "- ") -}}{{- $summary = "" -}}{{- end -}} {{- end -}} {{- return $summary -}}
